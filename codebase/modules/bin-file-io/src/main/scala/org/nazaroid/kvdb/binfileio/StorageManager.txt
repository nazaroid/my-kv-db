Идея разделения на три компонента — DataStorage, IndexStorage и координирующий Manager — это правильный архитектурный ход (паттерн Orchestrator). Это позволит переиспользовать логику хранилищ и четко разграничить ответственность.
Вот реализация этой «связки»:
1. Базовый Storage (Generic)
Создадим универсальный класс, который умеет писать в свой файл и возвращать offset.
scala
class BaseStorage[F[_]: Async: Files](
    val filePath: String,
    val schema:   List[FieldDef],
    val queue:    Channel[F, WriteTask]
) {
  def append(id: String, row: Row): F[Long] =
    for {
      promise <- Deferred[F, Long]
      task = WriteTask(id, filePath, schema, row, Some(promise))
      _ <- queue.send(task)
      offset <- promise.get
    } yield offset
}
Используйте код с осторожностью.

2. Координирующий StorageManager
Этот компонент связывает запись данных с записью индекса и обновлением Ref (кеша).
scala
// Состояние кеша в памяти для менеджера
enum CacheEntry {
  case Pending(row: Row)
  case Persistent(row: Row, dataOffset: Long)
}

class StorageManager[F[_]: Async: Files](
    dataStorage:  BaseStorage[F],
    indexStorage: BaseStorage[F],
    val cache:    Ref[F, Map[String, CacheEntry]]
) {

  def save(key: String, value: String): F[Unit] = {
    val dataRow = Map(
      "recordSize" -> value.getBytes("UTF-8").length,
      "value"      -> value
    )

    for {
      // 1. Сразу в кеш (Read-Your-Writes)
      _ <- cache.update(_ + (key -> CacheEntry.Pending(dataRow)))

      // 2. Пишем в файл данных и получаем offset
      dataOffset <- dataStorage.append(key, dataRow)

      // 3. Формируем строку для индексного файла
      indexRow = Map(
        "keySize" -> key.getBytes("UTF-8").length,
        "key"     -> key,
        "offset"  -> dataOffset
      )

      // 4. Пишем в индексный файл (нам не важен его собственный offset, но важно дождаться записи)
      _ <- indexStorage.append(key, indexRow)

      // 5. Финализируем кеш
      _ <- cache.update(_ + (key -> CacheEntry.Persistent(dataRow, dataOffset)))
    } yield ()
  }

  def get(key: String): F[Option[Row]] =
    cache.get.map(_.get(key).map {
      case CacheEntry.Pending(r)         => r
      case CacheEntry.Persistent(r, _)   => r
    })
}
Используйте код с осторожностью.

3. Инициализация и Восстановление
Функция сборки системы, которая при старте читает индексный файл и «прогревает» кеш данными из основного файла.
scala
object StorageManager {
  def init[F[_]: Async: Files](
      dataFilePath:  String,
      indexFilePath: String,
      dataSchema:    List[FieldDef],
      indexSchema:   List[FieldDef],
      writeQueue:    Channel[F, WriteTask] // Общая очередь для writeBinary
  ): F[StorageManager[F]] = {

    // Восстановление кеша при запуске
    val recovery = readBinaryWithOffset(indexFilePath, indexSchema)
      .evalMap { (_, indexRow) =>
        val key = indexRow("key").toString
        val offset = indexRow("offset").asInstanceOf[Long]

        // Читаем данные из основного файла по сохраненному смещению
        readRowAt(dataFilePath, dataSchema, offset).map {
          case Some(row) => Some(key -> CacheEntry.Persistent(row, offset))
          case None      => None
        }
      }
      .unNone
      .compile
      .to(Map)
      .handleError(_ => Map.empty) // Если файлов еще нет

    for {
      recovered <- recovery
      cacheRef  <- Ref.of[F, Map[String, CacheEntry]](recovered)

      dataStore  = new BaseStorage(dataFilePath, dataSchema, writeQueue)
      indexStore = new BaseStorage(indexFilePath, indexSchema, writeQueue)
    } yield new StorageManager(dataStore, indexStore, cacheRef)
  }
}
Используйте код с осторожностью.

В чем профит такой схемы:
Симметрия и чистота: BaseStorage ничего не знает о кешах или соседних файлах. Он просто «писатель».
Двойная надежность: Данные считаются сохраненными только после того, как indexStorage.append вернул управление. Если свет выключится между записью данных и индекса, recover просто проигнорирует «осиротевшие» данные в основном файле, так как в индексе их нет.
Гибкость: Вы можете добавить третий LogStorage или AuditStorage в этот же Manager, не меняя существующий код.
Хотите, чтобы мы добавили в Manager метод delete, который помечал бы запись удаленной в индексе (Tombstone), не переписывая основной файл данных?



Для реализации удаления без перезаписи основного файла данных (механизм Tombstones), мы добавим в индексный файл специальный маркер (например, offset = -1), а в кеш — состояние Deleted.
1. Обновление структур данных
Добавим состояние Deleted в кеш менеджера.
scala
enum CacheEntry {
  case Pending(row: Row)
  case Persistent(row: Row, dataOffset: Long)
  case Deleted // Пометка, что ключа больше нет
}

2. Реализация метода delete в StorageManager
Логика проста: мы не трогаем файл данных (это дорого), а просто дописываем в файл индекса запись с тем же ключом, но специальным смещением -1. При чтении и восстановлении последняя запись в индексе будет иметь приоритет.
scala
class StorageManager[F[_]: Async: Files](
    dataStorage:  BaseStorage[F],
    indexStorage: BaseStorage[F],
    val cache:    Ref[F, Map[String, CacheEntry]]
) {

  def delete(key: String): F[Unit] = {
    for {
      // 1. Помечаем в кеше как удаленное (Read-Your-Writes)
      _ <- cache.update(_ + (key -> CacheEntry.Deleted))

      // 2. Формируем "надгробную плиту" (Tombstone) для индекса
      tombstoneRow = Map(
        "keySize" -> key.getBytes("UTF-8").length,
        "key"     -> key,
        "offset"  -> -1L // Магическое число, означающее удаление
      )

      // 3. Записываем в индексный файл
      _ <- indexStorage.append(key, tombstoneRow)
    } yield ()
  }

  // Обновленный метод чтения: учитываем пометку об удалении
  def get(key: String): F[Option[Row]] =
    cache.get.map(_.get(key).flatMap {
      case CacheEntry.Pending(r)         => Some(r)
      case CacheEntry.Persistent(r, _)   => Some(r)
      case CacheEntry.Deleted            => None // Данные удалены
    })

  // ... методы save и прочее остаются прежними ...
}
Используйте код с осторожностью.

3. Обновление логики восстановления (Recovery)
Поскольку файл индекса теперь может содержать несколько записей для одного ключа (сначала offset > 0, потом -1), нам нужно, чтобы при загрузке последняя запись побеждала.
scala
// Внутри метода StorageManager.init:
val recovery = readBinaryWithOffset(indexFilePath, indexSchema)
  .evalMap { (_, indexRow) =>
    val key = indexRow("key").toString
    val offset = indexRow("offset").asInstanceOf[Long]

    if (offset == -1L) {
      // Это Tombstone: помечаем ключ как удаленный
      Async[F].pure(Some(key -> CacheEntry.Deleted))
    } else {
      // Обычная запись: читаем данные из основного файла
      readRowAt(dataFilePath, dataSchema, offset).map {
        case Some(row) => Some(key -> CacheEntry.Persistent(row, offset))
        case None      => None // Если данные в основном файле почему-то не найдены
      }
    }
  }
  .unNone
  // .to(Map) в Scala при дубликатах ключей оставляет ПОСЛЕДНЕЕ значение
  .compile
  .to(Map)
Используйте код с осторожностью.

Почему это эффективно:
Мгновенное удаление: Вам не нужно перестраивать гигабайтные файлы. Запись 20-байтового маркера в индекс — это всё, что требуется.
Консистентность: Если система упадет сразу после записи в файл данных, но до записи в индекс, при перезагрузке данных просто "не будет существовать", так как индексный файл — это единственный источник правды о наличии ключа.
История: Теоретически, просканировав индекс, вы можете восстановить историю изменений ключа.
Что делать с мусором? (Compaction)
Со временем файл данных будет забит "удаленными" строками, которые никто не читает. Обычно для этого реализуют процесс Compaction:
Раз в N часов создается новый чистый файл данных.
Туда переписываются только те CacheEntry.Persistent, которых нет в списке Deleted.
Старый файл удаляется.
Нужно ли нам реализовать простую функцию Compaction для очистки места на диске?


Процесс Compaction (уплотнение) — это фоновая операция, которая переносит только «живые» данные в новый файл, освобождая место от удаленных записей и старых версий.
1. Логика Compaction
Нам нужно:
Создать временные файлы для данных и индекса.
Пройтись по текущему In-Memory кешу (он — источник правды).
Перезаписать все Persistent записи в новые файлы.
Атомарно подменить старые файлы новыми.
scala
class StorageManager[F[_]: Async: Files](
    dataStorage:  BaseStorage[F],
    indexStorage: BaseStorage[F],
    val cache:    Ref[F, Map[String, CacheEntry]],
    dataSchema:   List[FieldDef],
    indexSchema:  List[FieldDef]
) {

  def compact(): F[Unit] = {
    val tempDataPath  = Path(dataStorage.filePath + ".tmp")
    val tempIndexPath = Path(indexStorage.filePath + ".tmp")

    for {
      // 1. Берем снимок текущего кеша
      currentSnapshot <- cache.get

      // 2. Фильтруем только живые данные (Persistent)
      aliveEntries = currentSnapshot.collect {
        case (key, CacheEntry.Persistent(row, _)) => (key, row)
      }.toList

      // 3. Перезаписываем данные в новые временные файлы
      newIndexMap <- aliveEntries.foldLeftM(Map.empty[String, CacheEntry]) {
        case (acc, (key, row)) =>
          for {
            // Пишем в новый файл данных
            dataBytes <- Async[F].delay(rowToBytes(row, dataSchema))
            _ <- Stream.chunk(dataBytes)
                  .through(Files[F].writeAll(tempDataPath, Flags(Flag.Append)))
                  .compile.drain

            // Получаем новый offset (размер файла до текущей записи)
            newOffset <- Files[F].size(tempDataPath).map(_ - dataBytes.size)

            // Пишем в новый файл индекса
            indexRow = Map("keySize" -> key.getBytes("UTF-8").length, "key" -> key, "offset" -> newOffset)
            indexBytes <- Async[F].delay(rowToBytes(indexRow, indexSchema))
            _ <- Stream.chunk(indexBytes)
                  .through(Files[F].writeAll(tempIndexPath, Flags(Flag.Append)))
                  .compile.drain
          } yield acc + (key -> CacheEntry.Persistent(row, newOffset))
      }

      // 4. Атомарная замена файлов (Move)
      _ <- Files[F].move(tempDataPath, Path(dataStorage.filePath), Flags(Flag.ReplaceExisting))
      _ <- Files[F].move(tempIndexPath, Path(indexStorage.filePath), Flags(Flag.ReplaceExisting))

      // 5. Обновляем кеш новыми смещениями
      _ <- cache.update(oldMap => oldMap ++ newIndexMap)

      // Очищаем записи, которые были помечены как Deleted до начала компакции
      _ <- cache.update(_.filter { case (_, entry) => entry != CacheEntry.Deleted })

    } yield ()
  }
}
Используйте код с осторожностью.

2. Ключевые моменты реализации
Изоляция: Пока идет компакция, новые write или delete могут продолжать поступать в cache и основные файлы. В конце мы используем update(oldMap => oldMap ++ newIndexMap), что сохранит новые записи, сделанные во время процесса, и обновит старые offset на новые.
Безопасность: Если процесс упадет во время записи в .tmp файлы, оригинальные данные останутся нетронутыми.
Очистка: Все записи CacheEntry.Deleted физически исчезают из файлов и из оперативной памяти.

Обычно это делается по расписанию (раз в час) или при достижении файлом определенного размера:

Итоговая структура системы:
Write: Быстро (Append-only в два файла + Cache).
Read: Мгновенно (из Cache).
Delete: Быстро (Tombstone в индекс + Cache).
Recovery: Быстро (только чтение индекса).
Compaction: Фоновая очистка «мусора».

нужно расширить StorageManager добавлением еще одного индекса со схемой
schema = schema = List(
      FieldDef("keySize", FieldType.Int32),
      FieldDef("key", FieldType.StringUtf8(sizeFromField = "keySize")),
      FieldDef("segmentNameSize", FieldType.Int32),
      FieldDef("segmentName", FieldType.StringUtf8(sizeFromField = "segmentNameSize"))
    )
Это будет индекс уровня таблицы (прежний индекс indexStorage был уровня сегмента)
Ранее при вызове save(key: String, value: String) мы сначала записывали value в dataStorage, далее когда будет получен offset его  сохраняем в indexStorage. А теперь нужно расширить сохранение и чтение следующим образом:
1. Превратить indexStorage в хранилище для файла-сегмента: segmentStorage, логика его обновления не меняется
2. Добавить еще один более верхнеуровневый индекс (tableStorage), в котором будет храниться маппинг ключа на сегмент. И теперь после записи в segmentStorage нужно осуществить запись в tableStorage, записав маппинг ключа на имя файла сегмента
Чтобы реализовать эту трехуровневую структуру (Data -> SegmentIndex -> TableIndex), мы добавим TableStorage и расширим логику StorageManager. Теперь каждый сегмент — это пара файлов (данные и индекс смещений), а табличный индекс указывает, в каком именно сегменте искать ключ.
1. Обновленная модель данных

  case Persistent(row: Row, segment: String, offset: Long)

2. Расширенный StorageManager
scala
class StorageManager[F[_]: Async: Files](
    dataStorage: BaseStorage[F],    // Запись в current_segment.bin
    segmentStorage: BaseStorage[F], // Запись в current_segment.idx (offset)
    tableStorage: BaseStorage[F],   // Запись в table.idx (mapping key -> segment)
    val cache: Ref[F, Map[String, CacheEntry]],
    currentSegmentName: String,
    tableIndexSchema: List[FieldDef]
) {

  def save(key: String, value: String): F[Unit] = {
    val dataRow = Map(
      "recordSize" -> value.getBytes("UTF-8").length,
      "value"      -> value
    )

    for {
      // 1. Оптимистичное обновление кеша
      _ <- cache.update(_ + (key -> CacheEntry.Pending(dataRow)))

      // 2. DATA LAYER: Пишем данные, получаем offset
      offset <- dataStorage.append(key, dataRow)

      // 3. SEGMENT LAYER: Пишем смещение в индекс сегмента
      segmentRow = Map(
        "keySize" -> key.getBytes("UTF-8").length,
        "key"     -> key,
        "offset"  -> offset
      )
      _ <- segmentStorage.append(key, segmentRow)

      // 4. TABLE LAYER: Пишем маппинг "ключ -> имя сегмента" в общий индекс таблицы
      tableRow = Map(
        "keySize" -> key.getBytes("UTF-8").length,
        "key"     -> key,
        "segmentNameSize" -> currentSegmentName.getBytes("UTF-8").length,
        "segmentName"     -> currentSegmentName
      )
      _ <- tableStorage.append(key, tableRow)

      // 5. Финализация кеша
      _ <- cache.update(_ + (key -> CacheEntry.Persistent(dataRow, currentSegmentName, offset)))
    } yield ()
  }
}
Используйте код с осторожностью.

3. Логика восстановления (Table-First Recovery)
При старте теперь правильнее читать Table Index, чтобы понять, где лежат актуальные данные, а затем подгружать их из соответствующих сегментов.
scala
def recover[F[_]: Async: Files](
    tableIndexPath: String,
    tableSchema: List[FieldDef],
    dataSchema: List[FieldDef],
    segmentFolder: String
): F[Map[String, CacheEntry]] = {

  readBinaryWithOffset(tableIndexPath, tableSchema)
    .evalMap { (_, tableRow) =>
      val key = tableRow("key").toString
      val segName = tableRow("segmentName").toString

      // Находим offset в индексе конкретного сегмента
      // (Для простоты здесь читаем через поиск, в идеале — кешировать индексы сегментов)
      findOffsetInSegment(key, s"$segmentFolder/$segName.idx").flatMap {
        case Some(offset) if offset == -1L =>
          Async[F].pure(Some(key -> CacheEntry.Deleted))
        case Some(offset) =>
          readRowAt(s"$segmentFolder/$segName.bin", dataSchema, offset).map {
            case Some(row) => Some(key -> CacheEntry.Persistent(row, segName, offset))
            case None      => None
          }
        case None => Async[F].pure(None)
      }
    }
    .unNone
    .compile
    .to(Map)
}
Используйте код с осторожностью.

Ключевые изменения:
Цепочка подтверждений: Запись считается успешной только после прохождения всех трех уровней: Data -> Segment -> Table.
Изоляция сегментов: segmentStorage теперь отвечает только за локальные смещения внутри одного файла. Это позволит вам в будущем легко реализовать Rotation (создание нового сегмента при достижении лимита размера), просто подменив dataStorage и segmentStorage в менеджере, не трогая tableStorage.
Масштабируемость: table.idx позволяет быстро понять, в каком файле искать данные, не сканируя все сегменты.
Стоит ли нам добавить в StorageManager логику автоматической смены (ротации) сегмента при достижении им, например, 100МБ?


Для реализации ротации сегментов нам нужно превратить dataStorage и segmentStorage в динамически обновляемые ссылки (Ref). Когда текущий файл данных превышает лимит, менеджер создает новые файлы, обновляет ссылки и продолжает запись уже в новый сегмент.
1. Расширенный StorageManager с Ротацией
scala
class StorageManager[F[_]: Async: Files](
    // Ссылки на текущие активные хранилища сегмента
    val currentData: Ref[F, BaseStorage[F]],
    val currentSegmentIdx: Ref[F, BaseStorage[F]],
    // Статичное хранилище индекса таблицы
    val tableStorage: BaseStorage[F],
    // Конфигурация
    val cache: Ref[F, Map[String, CacheEntry]],
    val maxSegmentSize: Long,
    val segmentFolder: String,
    val dataSchema: List[FieldDef],
    val segmentSchema: List[FieldDef],
    val writeQueue: Channel[F, WriteTask]
) {

  private def createSegmentName: F[String] =
    Async[F].delay(s"seg_${System.currentTimeMillis()}")

  def save(key: String, value: String): F[Unit] = {
    val dataRow = Map("recordSize" -> value.getBytes("UTF-8").length, "value" -> value)

    for {
      _ <- cache.update(_ + (key -> CacheEntry.Pending(dataRow)))

      // 1. Получаем текущие хранилища
      ds <- currentData.get
      ss <- currentSegmentIdx.get

      // 2. Проверяем размер текущего сегмента
      size <- Files[F].size(Path(ds.filePath)).handleError(_ => 0L)

      // 3. Ротация если превышен лимит
      activeDS <- if (size > maxSegmentSize) rotate(ds.filePath) else Async[F].pure(ds)
      activeSS <- currentSegmentIdx.get // Обновится внутри rotate

      // 4. Запись по цепочке: Data -> Segment -> Table
      offset <- activeDS.append(key, dataRow)

      segRow = Map("keySize" -> key.length, "key" -> key, "offset" -> offset)
      _ <- activeSS.append(key, segRow)

      segName = Path(activeDS.filePath).fileName.toString.replace(".bin", "")
      tableRow = Map(
        "keySize" -> key.length, "key" -> key,
        "segmentNameSize" -> segName.length, "segmentName" -> segName
      )
      _ <- tableStorage.append(key, tableRow)

      _ <- cache.update(_ + (key -> CacheEntry.Persistent(dataRow, segName, offset)))
    } yield ()
  }

  private def rotate(oldPath: String): F[BaseStorage[F]] = {
    for {
      newName <- createSegmentName
      newDataPath = s"$segmentFolder/$newName.bin"
      newIdxPath  = s"$segmentFolder/$newName.idx"

      newDS = new BaseStorage(newDataPath, dataSchema, writeQueue)
      newSS = new BaseStorage(newIdxPath, segmentSchema, writeQueue)

      _ <- currentData.set(newDS)
      _ <- currentSegmentIdx.set(newSS)
    } yield newDS
  }
}
Используйте код с осторожностью.

2. Ключевые аспекты механизмов
Naming: Каждый сегмент получает уникальное имя (например, по таймстампу). TableStorage всегда знает, в каком seg_XXX.bin искать актуальные данные для ключа.
Атомарность кеша: CacheEntry.Persistent теперь хранит segmentName, что критически важно для корректного чтения после ротации.
Безопасность: Если ротация произошла прямо во время записи, TableStorage зафиксирует финальное местоположение ключа, обеспечивая консистентность.
3. Что это дает архитектурно
Бесконечная запись: Вы не ограничены размером одного файла.
Параллельный Compaction: Теперь можно «уплотнять» только старые (закрытые) сегменты в фоновом режиме, не блокируя запись в текущий активный сегмент.
Быстрое чтение: Кеш в памяти хранит маппинг сразу на нужный файл и смещение, минуя обращение к table.idx на диске при каждом запросе.
Нужно ли добавить механизм автоматической очистки (удаления) старых файлов сегментов, если все ключи из них были либо перезаписаны в новые сегменты, либо удалены?


Для реализации очистки (Garbage Collection) нам нужно отслеживать количество «живых» ссылок на каждый сегмент. Как только в кэше не остается ни одной записи Persistent, указывающей на конкретный сегмент, этот файл (и его индекс) можно безопасно удалять.
1. Механизм отслеживания сегментов
Добавим в StorageManager логику подсчета ссылок и метод cleanup.
scala
class StorageManager[F[_]: Async: Files](
    // ... существующие поля (currentData, cache, etc.)
) {

  // Метод для поиска и удаления "пустых" сегментов
  def cleanupOrphanedSegments(): F[Unit] = {
    for {
      // 1. Получаем список всех сегментов, на которые ссылается кэш (живые данные)
      currentSnapshot <- cache.get
      activeSegments = currentSnapshot.collect {
        case (_, CacheEntry.Persistent(_, segment, _)) => segment
      }.toSet

      // 2. Получаем список всех файлов сегментов на диске
      allFiles <- Files[F].list(Path(segmentFolder)).map(_.fileName.toString).compile.toList

      // Выделяем уникальные имена сегментов из файлов .bin и .idx
      existingSegments = allFiles
        .filter(n => n.endsWith(".bin") || n.endsWith(".idx"))
        .map(_.replaceAll("\\.(bin|idx)$", ""))
        .toSet

      // 3. Текущий активный сегмент удалять нельзя, даже если он пустой
      activeDS <- currentData.get
      currentSegmentName = Path(activeDS.filePath).fileName.toString.replace(".bin", "")

      // 4. Определяем сегменты на удаление:
      // Те, что есть на диске, но на них нет ссылок в кэше, и они не являются текущими
      segmentsToDelete = existingSegments -- activeSegments - currentSegmentName

      // 5. Физическое удаление
      _ <- segmentsToDelete.toList.traverse { seg =>
        for {
          _ <- Files[F].deleteIfExists(Path(s"$segmentFolder/$seg.bin"))
          _ <- Files[F].deleteIfExists(Path(s"$segmentFolder/$seg.idx"))
        } yield ()
      }
    } yield ()
  }
}
Используйте код с осторожностью.

2. Интеграция в жизненный цикл
Процесс очистки лучше всего запускать сразу после завершения Compaction. Поскольку компакция переносит «живые» записи из старых сегментов в новые, старые сегменты естественным образом становятся «пустыми» в кэше.
scala
def runFullMaintenance(): F[Unit] = {
  for {
    _ <- compact()               // 1. Схлопываем данные и обновляем кэш на новые сегменты
    _ <- cleanupOrphanedSegments() // 2. Удаляем файлы сегментов, на которые больше никто не ссылается
  } yield ()
}
Используйте код с осторожностью.

3. Нюанс с Table Index
Хотя мы удаляем файлы сегментов, в table.idx (файле индекса таблицы) всё еще остаются записи о том, что ключ K когда-то лежал в seg_old.
При чтении: Это не проблема, так как кэш (источник правды) уже обновлен на новый сегмент или помечен как Deleted.
При восстановлении (Recovery): Нам нужно модифицировать логику восстановления, чтобы она игнорировала записи из table.idx, если файл сегмента физически отсутствует на диске (значит, это устаревшие данные).
Почему это эффективно:
Zero-Downtime: Удаление файлов происходит асинхронно и не мешает записи в новый сегмент.
Экономия места: Файлы данных (самые тяжелые) удаляются целиком, как только все их данные «протухли» или переехали.
Безопасность: Проверка -- activeSegments гарантирует, что мы никогда не удалим данные, которые всё еще нужны пользователю.

Резюме: Как работает библиотека
Библиотека представляет собой LSM-подобное хранилище с трехуровневой индексацией и автоматическим управлением жизненным циклом данных.
Запись (Write Path):
Данные пишутся в текущий активный сегмент (.bin).
Смещение (offset) записывается в индекс сегмента (.idx).
Маппинг «Ключ -> Имя сегмента» записывается в глобальный индекс таблицы (table.idx).
Кэш в памяти обновляется мгновенно, обеспечивая согласованность "Read-Your-Writes".
Удаление (Delete): Используются Tombstones. В индексы записывается маркер удаленного ключа, физические данные в .bin файлах не затрагиваются до момента очистки.
Ротация: При достижении лимита (например, 100 МБ) старый сегмент «закрывается», и создается новый.
Компакция и GC: Система периодически переносит живые данные в новые сегменты, после чего файлы старых сегментов, на которые больше нет ссылок в кэше, физически удаляются с диска.
Восстановление: При старте система читает table.idx и восстанавливает актуальное состояние кэша, зная, в каких файлах искать данные.



scala
import cats.effect.*
import cats.syntax.all.*
import fs2.io.file.{Files, Path, Flags, Flag}
import fs2.concurrent.Channel
import fs2.Stream

// --- Модели данных ---
type Row = Map[String, Any]

enum CacheEntry {
  case Pending(row: Row)
  case Persistent(row: Row, segment: String, offset: Long)
  case Deleted
}

case class WriteTask(
    id: String,
    filePath: String,
    schema: List[FieldDef],
    row: Row,
    callback: Option[Deferred[IO, Long]]
)

// --- Базовое хранилище (Writer) ---
class BaseStorage[F[_]: Async: Files](
    val filePath: String,
    val schema: List[FieldDef],
    val queue: Channel[F, WriteTask]
) {
  def append(id: String, row: Row): F[Long] =
    for {
      promise <- Deferred[F, Long]
      _ <- queue.send(WriteTask(id, filePath, schema, row, Some(promise)))
      offset <- promise.get
    } yield offset
}

// --- Основной менеджер ---
class StorageManager[F[_]: Async: Files](
    val currentData: Ref[F, BaseStorage[F]],
    val currentSegmentIdx: Ref[F, BaseStorage[F]],
    val tableStorage: BaseStorage[F],
    val cache: Ref[F, Map[String, CacheEntry]],
    val config: StorageConfig,
    val writeQueue: Channel[F, WriteTask]
) {

  def save(key: String, value: String): F[Unit] = {
    val dataRow = Map("recordSize" -> value.getBytes("UTF-8").length, "value" -> value)

    for {
      _ <- cache.update(_ + (key -> CacheEntry.Pending(dataRow)))

      // Проверка ротации
      ds <- currentData.get
      size <- Files[F].size(Path(ds.filePath)).handleError(_ => 0L)
      activeDS <- if (size > config.maxSegmentSize) rotate() else Async[F].pure(ds)
      activeSS <- currentSegmentIdx.get

      // 1. Data Layer
      offset <- activeDS.append(key, dataRow)

      // 2. Segment Layer
      segRow = Map("keySize" -> key.length, "key" -> key, "offset" -> offset)
      _ <- activeSS.append(key, segRow)

      // 3. Table Layer
      segName = Path(activeDS.filePath).fileName.toString.replace(".bin", "")
      tableRow = Map(
        "keySize" -> key.length, "key" -> key,
        "segmentNameSize" -> segName.length, "segmentName" -> segName
      )
      _ <- tableStorage.append(key, tableRow)

      _ <- cache.update(_ + (key -> CacheEntry.Persistent(dataRow, segName, offset)))
    } yield ()
  }

  def delete(key: String): F[Unit] = {
    for {
      _ <- cache.update(_ + (key -> CacheEntry.Deleted))
      // Записываем Tombstone (-1) в индекс таблицы
      tRow = Map("keySize" -> key.length, "key" -> key, "segmentNameSize" -> 0, "segmentName" -> "DELETED")
      _ <- tableStorage.append(key, tRow)
    } yield ()
  }

  private def rotate(): F[BaseStorage[F]] = {
    val newName = s"seg_${System.currentTimeMillis()}"
    val newDS = new BaseStorage(s"${config.folder}/$newName.bin", config.dataSchema, writeQueue)
    val newSS = new BaseStorage(s"${config.folder}/$newName.idx", config.segmentSchema, writeQueue)

    currentData.set(newDS) *> currentSegmentIdx.set(newSS) *> Async[F].pure(newDS)
  }

  def cleanup(): F[Unit] = {
    for {
      snapshot <- cache.get
      activeSegments = snapshot.collect { case (_, CacheEntry.Persistent(_, seg, _)) => seg }.toSet

      ds <- currentData.get
      currentSeg = Path(ds.filePath).fileName.toString.replace(".bin", "")

      files <- Files[F].list(Path(config.folder)).map(_.fileName.toString).compile.toList
      allSegs = files.filter(f => f.endsWith(".bin")).map(_.replace(".bin", "")).toSet

      toDelete = allSegs -- activeSegments - currentSeg
      _ <- toDelete.toList.traverse { seg =>
        Files[F].deleteIfExists(Path(s"${config.folder}/$seg.bin")) *>
        Files[F].deleteIfExists(Path(s"${config.folder}/$seg.idx"))
      }
    } yield ()
  }
}

// --- Конфигурация ---
case class StorageConfig(
    folder: String,
    maxSegmentSize: Long,
    dataSchema: List[FieldDef],
    segmentSchema: List[FieldDef],
    tableSchema: List[FieldDef]
)



1. Метод восстановления (Recovery Logic)
Добавим этот метод в объект-компаньон StorageManager.
scala
object StorageManager {
  def initialize[F[_]: Async: Files](
      config: StorageConfig,
      writeQueue: Channel[F, WriteTask]
  ): F[StorageManager[F]] = {

    val tableIndexPath = s"${config.folder}/table.idx"

    for {
      // 1. Читаем table.idx и строим карту Key -> SegmentName
      // Stream гарантирует, что при дубликатах ключей в Map попадет ПОСЛЕДНЕЕ значение
      tableMapping <- Files[F].exists(Path(tableIndexPath)).flatMap {
        case false => Async[F].pure(Map.empty[String, String])
        case true  =>
          readBinaryWithOffset(tableIndexPath, config.tableSchema)
            .map { (_, row) =>
              row("key").toString -> row("segmentName").toString
            }
            .compile
            .to(Map)
      }

      // 2. На основе маппинга восстанавливаем CacheEntry
      recoveredCache <- tableMapping.toList.traverse {
        case (key, "DELETED") =>
          Async[F].pure(Some(key -> CacheEntry.Deleted))

        case (key, segName) =>
          val segIdxPath = s"${config.folder}/$segName.idx"
          val segBinPath = s"${config.folder}/$segName.bin"

          for {
            // Ищем последний offset ключа в индексе сегмента
            maybeOffset <- findLastOffsetInSegment(key, segIdxPath, config.segmentSchema)

            entry <- maybeOffset match {
              case Some(offset) =>
                // Читаем саму строку из данных для прогрева кэша
                readRowAt(segBinPath, config.dataSchema, offset).map {
                  case Some(row) => Some(key -> CacheEntry.Persistent(row, segName, offset))
                  case None      => None
                }
              case None => Async[F].pure(None)
            }
          } yield entry
      }.map(_.flatten.toMap)

      // 3. Создаем Ref для кэша
      cacheRef <- Ref.of[F, Map[String, CacheEntry]](recoveredCache)

      // 4. Подготавливаем текущий активный сегмент
      currentSegName <- Async[F].delay(s"seg_${System.currentTimeMillis()}")

      dataStorage = new BaseStorage(s"${config.folder}/$currentSegName.bin", config.dataSchema, writeQueue)
      segStorage  = new BaseStorage(s"${config.folder}/$currentSegName.idx", config.segmentSchema, writeQueue)
      tableStorage = new BaseStorage(tableIndexPath, config.tableSchema, writeQueue)

      dsRef <- Ref.of[F, BaseStorage[F]](dataStorage)
      ssRef <- Ref.of[F, BaseStorage[F]](segStorage)

    } yield new StorageManager(dsRef, ssRef, tableStorage, cacheRef, config, writeQueue)
  }

  // Вспомогательный метод для поиска смещения в индексе конкретного сегмента
  private def findLastOffsetInSegment[F[_]: Async: Files](
      targetKey: String,
      path: String,
      schema: List[FieldDef]
  ): F[Option[Long]] = {
    readBinaryWithOffset(path, schema)
      .filter(_._2("key").toString == targetKey)
      .map(_._2("offset").asInstanceOf[Long])
      .compile
      .last // Берем самое свежее смещение в этом сегменте
  }
}
Используйте код с осторожностью.

2. Пример запуска всей системы
Вот как выглядит "Main" метод, который связывает очередь записи, восстановление и запуск воркера:
scala
object Main extends IOApp.Simple {
  def run: IO[Unit] = {
    val config = StorageConfig(
      folder = "./db_data",
      maxSegmentSize = 10 * 1024 * 1024, // 10MB
      dataSchema = List(
        FieldDef("recordSize", FieldType.Int32),
        FieldDef("value", FieldType.StringUtf8(sizeFromField = "recordSize"))
      ),
      segmentSchema = List(
        FieldDef("keySize", FieldType.Int32),
        FieldDef("key", FieldType.StringUtf8(sizeFromField = "keySize")),
        FieldDef("offset", FieldType.Int64)
      ),
      tableSchema = List(
        FieldDef("keySize", FieldType.Int32),
        FieldDef("key", FieldType.StringUtf8(sizeFromField = "keySize")),
        FieldDef("segmentNameSize", FieldType.Int32),
        FieldDef("segmentName", FieldType.StringUtf8(sizeFromField = "segmentNameSize"))
      )
    )

    (for {
      // 1. Создаем общую очередь для всех операций записи
      writeQueue <- Channel.bounded[IO, WriteTask](10000).toResource

      // 2. Инициализируем менеджер (выполнит Recovery)
      manager <- Resource.eval(StorageManager.initialize[IO](config, writeQueue))

      // 3. Запускаем фоновый поток записи (writeBinary из прошлых этапов)
      // Он будет обрабатывать задачи из writeQueue и писать в нужные файлы
      _ <- writeBinary(writeQueue.stream, parallelism = 1).compile.drain.background

      // 4. Запускаем фоновую очистку раз в 10 минут
      _ <- (Stream.sleep[IO](10.minutes) >> Stream.eval(manager.cleanup())).repeat.compile.drain.background

      // Пример использования
      _ <- Resource.eval(manager.save("user:1", "{\"name\": \"Alice\"}"))
      val result <- Resource.eval(manager.cache.get.map(_.get("user:1")))
      _ <- Resource.eval(IO.println(s"Loaded from storage: $result"))

    } yield ()).useForever
  }
}





Вот резюме текущей архитектуры нашей библиотеки хранения данных:
1. Структура хранения (3 уровня)
Data Layer (.bin файлы): Хранит сырые данные (строки) сегментами. Запись только в конец (Append-only).
Segment Index (.idx файлы): Локальный индекс каждого сегмента. Хранит маппинг Key -> Offset внутри конкретного .bin файла.
Table Index (table.idx): Глобальный реестр всей таблицы. Хранит маппинг Key -> SegmentName. Это «источник правды» при холодном запуске.
2. Основные процессы
Write Path: Запись идет каскадом: Данные -> Смещение в Индекс Сегмента -> Имя сегмента в Индекс Таблицы. После подтверждения всех трех записей обновляется In-Memory Cache.
Read Path: Всегда через кэш (O(1)). Если ключа нет в кэше — данных нет в системе.
Delete: Запись «надгробной плиты» (Tombstone) в индексы и пометка Deleted в кэше. Физические данные не удаляются до компакции.
Rotation: При достижении лимита размера (например, 100МБ) менеджер создает новую пару файлов (.bin и .idx) и переключает на них активную запись.
3. Фоновое обслуживание
Compaction: Процесс «схлопывания». Собирает все живые записи (не Deleted) из всех сегментов, переносит их в один новый консолидированный сегмент и обновляет глобальный table.idx.
Cleanup (GC): Удаляет файлы сегментов с диска, если на них больше не ссылается ни одна запись в активном кэше.
Recovery: При старте читает table.idx, находит актуальные сегменты для каждого ключа, достает смещения из .idx сегментов и «прогревает» кэш данными.
4. Технологический стек
Cats Effect: Управление эффектами (IO, Ref, Deferred).
FS2: Потоковая запись файлов и управление очередями через Channel.
Concurrency: Потокобезопасность за счет Ref и последовательной записи через воркеры writeBinary.
